-- 1. เคลียร์ตารางเก่าทิ้งให้สะอาด
DROP TABLE IF EXISTS comments CASCADE;
DROP TABLE IF EXISTS adoptions CASCADE;
DROP TABLE IF EXISTS cat_images CASCADE;
DROP TABLE IF EXISTS cats CASCADE;
DROP TABLE IF EXISTS breeds CASCADE;
DROP TABLE IF EXISTS colors CASCADE;
DROP TABLE IF EXISTS provinces CASCADE;
DROP TABLE IF EXISTS users CASCADE;

-- 2. สร้างโครงสร้างตารางใหม่ (เพิ่ม sort_order เพื่อจัดลำดับ "ไม่ทราบ" ไว้ล่างสุด)
CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    phone_number VARCHAR(20),
    password_hash VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE provinces (
    province_id SERIAL PRIMARY KEY,
    province_name VARCHAR(100) NOT NULL,
    sort_order INT DEFAULT 1
);

CREATE TABLE colors (
    color_id SERIAL PRIMARY KEY,
    color_name VARCHAR(50) NOT NULL,
    sort_order INT DEFAULT 1
);

CREATE TABLE breeds (
    breed_id SERIAL PRIMARY KEY,
    breed_name VARCHAR(100) NOT NULL,
    sort_order INT DEFAULT 1
);

CREATE TABLE cats (
    cat_id SERIAL PRIMARY KEY,
    owner_id INT REFERENCES users(user_id),
    cat_name VARCHAR(100) NOT NULL,
    gender VARCHAR(20),
    age_range VARCHAR(50),
    weight VARCHAR(50),
    vaccine_status VARCHAR(50),
    breed_id INT REFERENCES breeds(breed_id),
    color_id INT REFERENCES colors(color_id),
    province_id INT REFERENCES provinces(province_id),
    description TEXT,
    status VARCHAR(50) DEFAULT 'หาบ้าน',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE cat_images (
    image_id SERIAL PRIMARY KEY,
    cat_id INT REFERENCES cats(cat_id) ON DELETE CASCADE,
    image_path TEXT NOT NULL,
    is_cover BOOLEAN DEFAULT FALSE
);

CREATE TABLE adoptions (
    adoption_id SERIAL PRIMARY KEY,
    cat_id INT REFERENCES cats(cat_id) ON DELETE CASCADE,
    adopter_id INT REFERENCES users(user_id),
    status VARCHAR(50) DEFAULT 'Pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE comments (
    comment_id SERIAL PRIMARY KEY,
    cat_id INT REFERENCES cats(cat_id) ON DELETE CASCADE,
    user_id INT REFERENCES users(user_id),
    message TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
